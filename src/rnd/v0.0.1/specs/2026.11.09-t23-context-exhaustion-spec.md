## 1. Why this item changed shape in the session

T23 was a *measure* item: "what the engine does when its window fills; the context-pressure threshold that triggers a memento respin and writes `watch.pressure`". The first draft of the narrative proposed a capture that fills a one-million-token window on each engine and model. Rick ruled that a research step precedes any capture that burns context, since the behaviour might already be documented. It was. The design below rests on that record and the capture shrank to a confirmation.

## 5. The detector (session task)

```python
# pressure.py — evaluated by the session task on every normalized `usage` event. No timer, no poller.
from dataclasses import dataclass

@dataclass
class Pressure:
    occupancy_tokens : int | None
    window_tokens    : int | None
    ceiling_tokens   : int | None
    ratio            : float | None
    status           : str            # ok | notice | over_budget | unknown

def occupancy_from_usage( engine, usage ):
    """Ensures: the prompt size of the newest model call, never a cumulative figure (Jetski family result.usage is skipped by the adapter)."""
    if engine == "claude":
        return usage.input + usage.cache_creation + usage.cache_read
    return usage.input_tokens + usage.cache_read_tokens            # Jetski family agent_response DONE step

def lines_for( ceiling, config, manifest ):
    cls = "1m" if ceiling >= config.class_boundary_tokens else "200k"
    notice = config.notice_ratio[ cls ]
    respin = manifest.pressure.respin_ratio or config.respin_ratio[ cls ]
    return notice, respin

def assess( session, usage, now ):
    row = session.config.context_windows.get( ( session.engine, session.model ) )
    if row is None:
        return Pressure( None, None, None, None, "unknown" )            # narrated once at spawn as pressure.window_unknown
    occ    = occupancy_from_usage( session.engine, usage )
    ratio  = occ / row.ceiling_tokens
    notice, respin = lines_for( row.ceiling_tokens, session.config.pressure, session.manifest )
    status = "over_budget" if ratio >= respin else "notice" if ratio >= notice else "ok"
    p = Pressure( occ, row.window_tokens, row.ceiling_tokens, ratio, status )
    session.registry.pressure = { "ratio": ratio, "status": status, "measured_at": now() }

    # implicit compaction: a drop of more than half with no kernel respawn between two model calls
    prev = session.last_occupancy
    if prev and not session.respawned_since( prev ) and occ < prev * ( 1 - session.config.pressure.compaction_drop_fraction ):
        on_compacted( session, before=prev, after=occ, explicit=False )
    session.last_occupancy = occ

    # one record per upward crossing per line; the respawn resets session.crossed
    for line, threshold in ( ( "notice", notice ), ( "respin", respin ) ):
        if ratio >= threshold and line not in session.crossed:
            session.crossed.add( line )
            action = "none"
            if line == "respin":
                action = session.manifest.pressure.action                      # respin | advise
            log( "watch.pressure", persona_id=session.persona_id, turn_id=session.turn_id, occupancy_tokens=occ,
                 window_tokens=row.window_tokens, ceiling_tokens=row.ceiling_tokens, ratio=round( ratio, 4 ),
                 line=line, action=action, model=session.model, source=usage.source )
            if line == "respin":
                if action == "respin":
                    session.kernel.schedule( respin, session.persona, trigger="pressure" )   # D54 handshake; waits for turn end
                session.kernel.narrate_chain( session.persona, type="info",
                    text=f"{session.persona.name} is at {ratio:.0%} of its {row.ceiling_tokens:,}-token ceiling; "
                         + ( "re-spinning at turn end." if action == "respin" else "re-spin advised." ) )
    return p

def on_checkpoint_step( session, step ):
    """The adapter lifts a Jetski-family `checkpoint` step (or a Claude compact_boundary) here."""
    on_compacted( session, before=session.last_occupancy, after=None, explicit=True, stall_seconds=step.duration_seconds )

def on_compacted( session, before, after, explicit, stall_seconds=None ):
    log( "watch.compacted", persona_id=session.persona_id, turn_id=session.turn_id,
         before_tokens=before, after_tokens=after, explicit=explicit, stall_seconds=stall_seconds )
    session.crossed.clear()                                                      # the lines are re-armed; the next climb is a new crossing
    session.kernel.inbox.write_kernel_block( session.persona, "memento",
        "Your engine compacted its context. At your next safe checkpoint, write your memento (memento_io.py write --slot io).",
        urgent=False )
    session.kernel.narrate_chain( session.persona, type="info", text=f"{session.persona.name}'s engine compacted its context." )
    if session.manifest.pressure.compaction_action == "interrupt":              # only for a model C19 shows to fail at the edge
        session.kernel.interrupt( session.persona, reason="pressure_compaction" )
```

`respin.py` (T18 spec Section 9) gains one keyword: `respin( kernel, persona, now, trigger="operator" )`; `respin.requested`, `respin.rehydrated` and `respin.without_memento` carry `trigger: operator | pressure | persona`. The pressure trigger waits for the **current** turn to end before writing the request block only on engines without mid-turn input (`accepts_input_mid_turn: no`); on Claude the block is written at once and absorbed.

```python
"""
manager_advice.py: what the kernel tells a persona's manager on its own account, to whom and how loudly.

Ruled 2026-11-20 (row 107 of "Outstanding actions", under D57, context exhaustion):
  1   a line to a manager is a `kernel` block in the manager's inbox, never urgent: it never interrupts a turn;
      a persona with no manager has no block
  2   the manager receives its block, and nobody above the manager; the human receives one card for every event,
      whose priority and type are suggestions that the receiving client renders as it is configured (Rick)
  3   `alert` (card `priority: high`, `type: alert`): a held tool call, a process killed for memory, context pressure
      under `advise`, unmerged commits at dismissal (Rick: the manager reassigns them);
      `info` (card `priority: low`, `type: info`): context pressure under `respin`, a compaction, a day's hibernation.
      `info` is a notification type Lupin gains (Rick); the head word `report` is retired for these lines
  4   a compaction sends the manager a line
  5   the seven head words and sentences below
  6   the held call and the memory kill are ruled with the other four
  7   the one call is `advise_manager( persona, level, word, body )`

The block's grammar is the one ruled the same morning for a deferred wake (row 104):
`[kernel HH:MM:SS <alert|info> <word> <id8>]`, then the body.

Dependency-free.
"""
import collections

ALERT = "alert"
INFO  = "info"

CARD_LEVELS = { ALERT: ( "high", "alert" ), INFO: ( "low", "info" ) }        # ( priority, type ) suggested on the human's card

PRESSURE   = "pressure"
COMPACTED  = "compacted"
HIBERNATED = "hibernated"
UNMERGED   = "unmerged"
HELD       = "held"
KILLED     = "killed"

RESPIN    = "respin"
ADVISE    = "advise"
INTERRUPT = "interrupt"

Advice = collections.namedtuple( "Advice", "level word body" )

def check_level( level ):
    """Raises: ValueError for a level that is neither `alert` nor `info`."""
    if level not in CARD_LEVELS:
        raise ValueError( f"a line to a manager is an alert or an info, not {level!r}" )

def build_pressure_advice( name, ratio, ceiling_tokens, action ):
    """Ensures: the line for a persona that crossed its re-spin line. `action` is the manifest's `pressure.action`."""
    if action not in ( RESPIN, ADVISE ):
        raise ValueError( f"pressure.action is respin or advise, not {action!r}" )
    where = f"{name} is at {ratio:.0%} of its {ceiling_tokens:,}-token ceiling."
    if action == RESPIN:
        return Advice( INFO, PRESSURE, where + " It re-spins when its turn ends." )
    return Advice( ALERT, PRESSURE, where + " Nothing re-spins it: its settings say advise only." )

def build_compaction_advice( name ):
    """Ensures: the line for a persona whose engine compacted its own context."""
    return Advice( INFO, COMPACTED, f"{name}'s engine compacted its context. It has been asked to write its memento." )

def build_hibernation_advice( name, hours ):
    """Ensures: the line for a persona hibernated past `kernel fleet idle advise hours`; `hours` is that key's value."""
    return Advice( INFO, HIBERNATED, f"{name} has been hibernated for {hours} hours. It keeps its place in the fleet until it is dismissed." )

def build_unmerged_advice( name, commits, branch ):
    """Ensures: the line for a dismissal whose branch probe found commits the working line lacks."""
    if commits < 1:
        raise ValueError( "the branch probe advises only when it found a commit" )
    counted = "1 commit" if commits == 1 else f"{commits} commits"
    return Advice( ALERT, UNMERGED, f"{name} was dismissed with {counted} its working line lacks, on {branch}. The branch is kept. "
                                    "Reassign it: someone must merge it or finish its work." )

def build_held_call_advice( name, tool, age_seconds, action ):
    """Ensures: the line for a tool call held past its threshold. `action` is the manifest's `watch.held_turn_action`."""
    if action not in ( ADVISE, INTERRUPT ):
        raise ValueError( f"watch.held_turn_action is advise or interrupt, not {action!r}" )
    what = f"{name}'s call to {tool} has been open for {int( age_seconds )} seconds with nothing from the engine."
    after = " It has been interrupted." if action == INTERRUPT else " It is not interrupted: its settings say advise only."
    return Advice( ALERT, HELD, what + after )

def build_kill_advice( name ):
    """Ensures: the line for a persona whose process was killed for memory."""
    return Advice( ALERT, KILLED, f"{name}'s process was killed for memory. Its row is kept, and it restarts at its next message." )

def build_manager_block( hhmmss, persona_id, level, word, body ):
    """Ensures: the `kernel` block written to the manager's inbox, not urgent. `persona_id` is the persona the line is about."""
    check_level( level )
    return f"[kernel {hhmmss} {level} {word} {persona_id}]\n{body}"

def build_human_card( level, body ):
    """Ensures: the one card to the human; its priority and type are suggestions to the client, from the line's level."""
    check_level( level )
    priority, kind = CARD_LEVELS[ level ]
    return { "text": body, "priority": priority, "type": kind }

def advise_manager( persona, level, word, body, *, hhmmss, write_block, post_card ):
    """
    Tell a persona's manager, once, something the kernel has to say about that persona.

    Requires: `persona` has `persona_id` and `manager_id` (None for a persona with no manager);
              `write_block( manager_id, block )` queues a block that is not urgent in that manager's inbox;
              `post_card( card )` posts one card to the human.
    Ensures:  one block to the manager when there is one; one card to the human every time;
              nothing is written above the manager.
    Returns:  whether a manager was told.
    """
    card = build_human_card( level, body )                                   # checks the level before anything is sent
    has_manager = persona.manager_id is not None
    if has_manager:
        write_block( persona.manager_id, build_manager_block( hhmmss, persona.persona_id, level, word, body ) )
    post_card( card )
    return has_manager
```

"""
deferred_wake.py: what a deferred wake says, to whom and how loudly, by its cause.

Ruled 2026-11-20 (row 104 of "Outstanding actions", under D62 part 10):
  2   the manager's line has one form: whose wake, why, that the message is kept, what ends the wait; it asks for nothing
  3   by cause, without interrupting the manager: a broken settings file is `priority: high`, `type: alert`, and the manager's
      `kernel` block is headed `alert deferred`; no room on the host is `priority: low`, `type: info`, and the block is headed
      `info deferred` (amended the same day, row 107: `report` is `info`); the human gets one card for every deferred wake,
      at the level below, a suggestion to the client; the settings file's own spoken line, once for each refused state of
      the file, is `priority: urgent`, `type: alert` (Rick, row 107)

Worded for the two causes Kernel-1 builds. The two sign-in and quota causes (`engine_login`, `engine_quota`) have no ruled
words yet and are refused here, so that nothing invents them.

Dependency-free.
"""
CONFIGURATION = "configuration"
HOST_CAPACITY = "host_capacity"

NOTICE_LEVELS = { CONFIGURATION: ( "high", "alert" ), HOST_CAPACITY: ( "low", "info" ) }
BLOCK_WORDS   = { CONFIGURATION: "alert deferred", HOST_CAPACITY: "info deferred" }
SETTINGS_FILE_NOTICE_LEVEL = ( "urgent", "alert" )      # the line `announce()` returns, once for each refused state of the file
CAUSES        = { CONFIGURATION: "the kernel's settings file has a bad key, {key}", HOST_CAPACITY: "the host has no room for it" }
REMEDIES      = { CONFIGURATION: "the file is corrected", HOST_CAPACITY: "room is freed" }

def check_reason( reason ):
    """Raises: ValueError for a cause whose words are not ruled."""
    if reason not in NOTICE_LEVELS:
        raise ValueError( f"no line is ruled for a wake deferred by {reason!r}" )

def get_notice_level( reason ):
    """Ensures: ( priority, type ) suggested on the human's card for one deferred wake."""
    check_reason( reason )
    return NOTICE_LEVELS[ reason ]

def build_manager_sentence( name, reason, key=None ):
    """Ensures: the body of the manager's block, two sentences. `key` is the key at fault, for `configuration` alone."""
    check_reason( reason )
    if ( reason == CONFIGURATION ) != ( key is not None ):
        raise ValueError( "a key is named for a broken settings file and for nothing else" )
    return f"{name}'s wake is deferred: {CAUSES[ reason ].format( key=key )}. Its message is kept, and it wakes when {REMEDIES[ reason ]}."

def build_manager_block( hhmmss, persona_id, name, reason, key=None ):
    """Ensures: the `kernel` block written to the manager's inbox, not urgent: it never interrupts a turn."""
    return f"[kernel {hhmmss} {BLOCK_WORDS[ reason ]} {persona_id}]\n" + build_manager_sentence( name, reason, key )

def build_human_card( name, reason, key=None ):
    """Ensures: the human's one card for a deferred wake, sent whether or not the persona has a manager."""
    priority, kind = get_notice_level( reason )
    return { "text": build_manager_sentence( name, reason, key ), "priority": priority, "type": kind }

`manager_advice.py`:

```python
"""
manager_advice.py: what the kernel tells a persona's manager on its own account, to whom and how loudly.

Ruled 2026-11-20 (row 107 of "Outstanding actions", under D57, context exhaustion):
  1   a line to a manager is a `kernel` block in the manager's inbox, never urgent: it never interrupts a turn;
      a persona with no manager has no block
  2   the manager receives its block, and nobody above the manager; the human receives one card for every event,
      whose priority and type are suggestions that the receiving client renders as it is configured (Rick)
  3   `alert` (card `priority: high`, `type: alert`): a held tool call, a process killed for memory, context pressure
      under `advise`, unmerged commits at dismissal (Rick: the manager reassigns them);
      `info` (card `priority: low`, `type: info`): context pressure under `respin`, a compaction, a day's hibernation.
      `info` is a notification type Lupin gains (Rick); the head word `report` is retired for these lines
  4   a compaction sends the manager a line
  5   the seven head words and sentences below
  6   the held call and the memory kill are ruled with the other four
  7   the one call is `advise_manager( persona, level, word, body )`

Ruled 2026-11-21 (row 108 of "Outstanding actions", under D57):
  1   a deferred wake is an eighth line: `deferred_wake.build_deferral_advice` builds it and `advise_manager` sends it
  2   the card's three fields are Lupin's own names: `message`, `priority`, `type`
  6   a block or a card that cannot be delivered never reaches the caller: each writer is tried on its own, a failure
      is logged as `advice.undelivered { persona_id, level, word, channel, error }`, and nothing is retried
  8   the word agrees with the number: "1 hour", "1 second", as "1 commit"
  10  only an `alert` wakes a manager that is itself hibernated; an `info` block waits in its inbox for its next wake:
      `write_block( manager_id, block, wakes )`

The block's grammar is the one ruled the same morning for a deferred wake (row 104):
`[kernel HH:MM:SS <alert|info> <word> <id8>]`, then the body.

Dependency-free.
"""
import collections

ALERT = "alert"
INFO  = "info"

CARD_LEVELS = { ALERT: ( "high", "alert" ), INFO: ( "low", "info" ) }        # ( priority, type ) suggested on the human's card

PRESSURE   = "pressure"
COMPACTED  = "compacted"
HIBERNATED = "hibernated"
UNMERGED   = "unmerged"
HELD       = "held"
KILLED     = "killed"
DEFERRED   = "deferred"                                                      # built by deferred_wake.py

RESPIN    = "respin"
ADVISE    = "advise"
INTERRUPT = "interrupt"

UNDELIVERED = "advice.undelivered"                                           # the record of a block or a card that failed

Advice = collections.namedtuple( "Advice", "level word body" )

def check_level( level ):
    """Raises: ValueError for a level that is neither `alert` nor `info`."""
    if level not in CARD_LEVELS:
        raise ValueError( f"a line to a manager is an alert or an info, not {level!r}" )

def build_count( number, unit ):
    """Ensures: the number and its unit, the unit singular for exactly one: "1 hour", "24 hours"."""
    return f"{number} {unit}" if number == 1 else f"{number} {unit}s"

def build_pressure_advice( name, ratio, ceiling_tokens, action ):
    """Ensures: the line for a persona that crossed its re-spin line. `action` is the manifest's `pressure.action`."""
    if action not in ( RESPIN, ADVISE ):
        raise ValueError( f"pressure.action is respin or advise, not {action!r}" )
    where = f"{name} is at {ratio:.0%} of its {ceiling_tokens:,}-token ceiling."
    if action == RESPIN:
        return Advice( INFO, PRESSURE, where + " It re-spins when its turn ends." )
    return Advice( ALERT, PRESSURE, where + " Nothing re-spins it: its settings say advise only." )

def build_compaction_advice( name ):
    """Ensures: the line for a persona whose engine compacted its own context."""
    return Advice( INFO, COMPACTED, f"{name}'s engine compacted its context. It has been asked to write its memento." )

def build_hibernation_advice( name, hours ):
    """Ensures: the line for a persona hibernated past `kernel fleet idle advise hours`; `hours` is that key's value."""
    return Advice( INFO, HIBERNATED, f"{name} has been hibernated for {build_count( hours, 'hour' )}. It keeps its place in the fleet until it is dismissed." )

def build_unmerged_advice( name, commits, branch ):
    """Ensures: the line for a dismissal whose branch probe found commits the working line lacks."""
    if commits < 1:
        raise ValueError( "the branch probe advises only when it found a commit" )
    return Advice( ALERT, UNMERGED, f"{name} was dismissed with {build_count( commits, 'commit' )} its working line lacks, on {branch}. The branch is kept. "
                                    "Reassign it: someone must merge it or finish its work." )

def build_held_call_advice( name, tool, age_seconds, action ):
    """Ensures: the line for a tool call held past its threshold. `action` is the manifest's `watch.held_turn_action`."""
    if action not in ( ADVISE, INTERRUPT ):
        raise ValueError( f"watch.held_turn_action is advise or interrupt, not {action!r}" )
    what = f"{name}'s call to {tool} has been open for {build_count( int( age_seconds ), 'second' )} with nothing from the engine."
    after = " It has been interrupted." if action == INTERRUPT else " It is not interrupted: its settings say advise only."
    return Advice( ALERT, HELD, what + after )

def build_kill_advice( name ):
    """Ensures: the line for a persona whose process was killed for memory."""
    return Advice( ALERT, KILLED, f"{name}'s process was killed for memory. Its row is kept, and it restarts at its next message." )

def build_manager_block( hhmmss, persona_id, level, word, body ):
    """Ensures: the `kernel` block written to the manager's inbox, not urgent. `persona_id` is the persona the line is about."""
    check_level( level )
    return f"[kernel {hhmmss} {level} {word} {persona_id}]\n{body}"

def build_human_card( level, body ):
    """Ensures: the one card to the human; its priority and type are suggestions to the client, from the line's level."""
    check_level( level )
    priority, kind = CARD_LEVELS[ level ]
    return { "message": body, "priority": priority, "type": kind }

def advise_manager( persona, level, word, body, *, hhmmss, write_block, post_card, write_record ):
    """
    Tell a persona's manager, once, something the kernel has to say about that persona.

    Requires: `persona` has `persona_id` and `manager_id` (None for a persona with no manager);
              `write_block( manager_id, block, wakes )` queues a block that is not urgent in that manager's inbox,
              and asks for a hibernated manager's wake only when `wakes` is true;
              `post_card( card )` posts one card to the human;
              `write_record( name, fields )` writes one record to the kernel's log.
    Ensures:  one block to the manager when there is one, waking it only for an alert; one card to the human
              every time; nothing is written above the manager; each writer is tried on its own, a failure of
              either is logged as `advice.undelivered` and never raised, and nothing is retried.
    Raises:   ValueError for an unknown level, before anything is sent.
    Returns:  whether a manager was told, that is, whether its block was written.
    """
    card = build_human_card( level, body )                                   # checks the level before anything is sent
    told = False
    if persona.manager_id is not None:
        block = build_manager_block( hhmmss, persona.persona_id, level, word, body )
        try:
            write_block( persona.manager_id, block, level == ALERT )
            told = True
        except Exception as error:
            write_record( UNDELIVERED, build_undelivered_fields( persona.persona_id, level, word, "block", error ) )
    try:
        post_card( card )
    except Exception as error:
        write_record( UNDELIVERED, build_undelivered_fields( persona.persona_id, level, word, "card", error ) )
    return told

def build_undelivered_fields( persona_id, level, word, channel, error ):
    """Ensures: the fields of `advice.undelivered`; `channel` is `block` or `card`, `error` the failure's own text."""
    return { "persona_id": persona_id, "level": level, "word": word, "channel": channel, "error": str( error ) }
```

Ruled 2026-11-20 (row 104 of "Outstanding actions", under D62 part 10):
  2   the manager's line has one form: whose wake, why, that the message is kept, what ends the wait; it asks for nothing
  3   by cause, without interrupting the manager: a broken settings file is an `alert` (the human's card `priority: high`,

      `info deferred`); the human gets one card for every deferred wake; the settings file's own spoken line, once for
      each refused state of the file, is `priority: urgent`, `type: alert` (Rick, row 107)

  1   the deferred wake is an eighth line of `manager_advice.py`: this module returns an `Advice` and sends nothing
  3   a cause whose words are not ruled is a ValueError on every path; so is a key that is missing, empty or blank

  1   a lapsed sign-in: "its engine's sign-in has lapsed", ended when "the sign-in is restored"; no parameter
  2   it is an `alert`, as a broken settings file is: only a person ends it
  3   the lapse's own card and spoken line, once for each lapse and at each reminder, are `priority: urgent`, `type: alert`
  4   a wake deferred by a blocked pool sends its block and its card like every other deferred wake
  5   an exhausted allowance: "the <name> token buffet is exhausted until <time>", ended when "the buffet refills";
      "token buffet" is the word for a pool wherever a person reads it (Rick); records and keys keep `pool`
  6   the time is written `3:42 PM Thursday`, in `app timezone`, the zone not named
  7   it is an `info`, as no room on the host is: time ends it
  8   the pool's own cards: blocked `priority: high`, `type: alert`; reopened `priority: low`, `type: info`
  and, as findings while recording: the names below; the pool's name in words is passed in by the caller

Ruled 2026-11-23 (row 21, under D68):
  8   a reset six or more days away carries its date, `3:42 PM Thursday, December 3`; nearer, the form of row 106 stands.
      The weekday alone is ambiguous at that distance, so the present time is passed in beside the reset time

All four causes of a deferred wake now have words. A cause this module does not know is still refused.

Depends on `manager_advice.py` alone.
"""
from manager_advice import ALERT, DEFERRED, INFO, Advice

CONFIGURATION = "configuration"
HOST_CAPACITY = "host_capacity"
ENGINE_LOGIN  = "engine_login"
ENGINE_QUOTA  = "engine_quota"

LEVELS   = { CONFIGURATION: ALERT, HOST_CAPACITY: INFO, ENGINE_LOGIN: ALERT, ENGINE_QUOTA: INFO }
CAUSES   = { CONFIGURATION: "the kernel's settings file has a bad key, {key}",
             HOST_CAPACITY: "the host has no room for it",
             ENGINE_LOGIN:  "its engine's sign-in has lapsed",
             ENGINE_QUOTA:  "the {pool_name} token buffet is exhausted until {reset_time}" }
REMEDIES = { CONFIGURATION: "the file is corrected", HOST_CAPACITY: "room is freed",
             ENGINE_LOGIN:  "the sign-in is restored", ENGINE_QUOTA: "the buffet refills" }

# The ( priority, type ) of each condition's own line to the human, sent apart from any persona it stops.
SETTINGS_FILE_NOTICE_LEVEL = ( "urgent", "alert" )      # the line `announce()` returns, once for each refused state of the file
SIGN_IN_NOTICE_LEVEL       = ( "urgent", "alert" )      # the lapse's card and spoken line, and each reminder of it
POOL_BLOCKED_NOTICE_LEVEL  = ( "high", "alert" )        # one card for each pool and block
POOL_REOPENED_NOTICE_LEVEL = ( "low", "info" )          # one card when the pool reopens

WEEKDAYS = ( "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday" )
MONTHS   = ( "January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December" )
DATED_FROM_DAYS = 6     # a reset this many days away or more carries its date (row 21, ruling 8)

def check_reason( reason ):
    """Raises: ValueError for a cause whose words are not ruled."""
    if reason not in LEVELS:
        raise ValueError( f"no line is ruled for a wake deferred by {reason!r}" )

def format_reset_time( moment, now ):
    """
    Requires: `moment` and `now` are datetimes already in `app timezone`, both aware or both naive.
    Ensures:  the time as a line states it, "3:42 PM Thursday": twelve-hour, no leading zero, the weekday, no zone;
              with its date, "3:42 PM Thursday, December 3", when it is six or more days after `now`.
    """
    hour  = moment.hour % 12 or 12
    half  = "AM" if moment.hour < 12 else "PM"
    words = f"{hour}:{moment.minute:02d} {half} {WEEKDAYS[ moment.weekday() ]}"
    if ( moment - now ).total_seconds() >= DATED_FROM_DAYS * 86400:
        words += f", {MONTHS[ moment.month - 1 ]} {moment.day}"
    return words

def is_named( value ):
    """Ensures: whether a string parameter was given and is not blank."""
    return value is not None and value.strip() != ""

def build_manager_sentence( name, reason, key=None, pool_name=None, reset_time=None, now=None ):
    """
    Ensures: the body of the manager's block and the human's card, two sentences.
             `key` is the key at fault, for `configuration` alone;
             `pool_name` is the pool's vendor or family name in words ("Claude"), `reset_time` the datetime, in `app timezone`,
             until which it is blocked, and `now` the present time in the same zone, all three for `engine_quota` alone.
    Raises:  ValueError for a cause whose words are not ruled, and for a parameter missing, blank or named for another cause.
    """
    check_reason( reason )
    if reason == CONFIGURATION and not is_named( key ) or reason != CONFIGURATION and key is not None:
        raise ValueError( "a key is named for a broken settings file and for nothing else" )
    quota_named = is_named( pool_name ) and reset_time is not None and now is not None
    if reason == ENGINE_QUOTA and not quota_named or reason != ENGINE_QUOTA and ( pool_name is not None or reset_time is not None or now is not None ):
        raise ValueError( "a pool, its reset time and the present time are named for an exhausted allowance and for nothing else" )
    time_words = format_reset_time( reset_time, now ) if reason == ENGINE_QUOTA else None
    cause = CAUSES[ reason ].format( key=key, pool_name=pool_name, reset_time=time_words )
    return f"{name}'s wake is deferred: {cause}. Its message is kept, and it wakes when {REMEDIES[ reason ]}."

def build_deferral_advice( name, reason, key=None, pool_name=None, reset_time=None, now=None ):
    """Ensures: the line for one deferred wake, which `advise_manager` sends as it sends the other seven."""
    check_reason( reason )
    return Advice( LEVELS[ reason ], DEFERRED, build_manager_sentence( name, reason, key, pool_name, reset_time, now ) )
